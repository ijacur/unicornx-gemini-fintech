⚡ 1. O'zgaruvchan Sakrash (Variable Jump)

Tugma bosib turilsa standart gravitatsiya qo'llaniladi. Agar o'yinchi tugmani erta qo'yib yuborsa, LOW_JUMP_GRAVITY_MULT (2.5x) faollashib, o'yinchining ko'tarilishini keskin to'xtatadi. Tushishda esa FALL_GRAVITY_MULT (1.8x) og'irlik hissini beradi.

📦 2. O'qlarga Bo'lingan AABB (Axis-Separation)

Kolliziyalar X va Y o'qlari bo'yicha mustaqil yechiladi. X o'qi surilib devor to'qnashuvi hal qilinadi, so'ngra Y o'qi surilib pol va shift to'qnashuvi yechiladi. Bu burchaklarda tiqilib qolishning (corner snagging) 100% oldini oladi.

🤖 3. Dushman Zondlash AI (Edge Detection)

Goomba uslubidagi dushman oldinga va pastga raycast/overlap nuqtalarini yo'naltiradi. Agar oldinda to'siq (devor) bo'lsa yoki pastida blok yo'q (jarlik) bo'lsa, u darhol harakat yo'nalishini 180° ga o'zgartiradi.

📷 4. Monoton Bir Tomonlama Kamera

Kamera o'yinchi ortidan silliq (lerp) kuzatadi, biroq maxScrollX = max(maxScrollX, camX) sharti orqali uning orqaga qaytishi taqiqlanadi. O'yinchi ekranning chap chegarasiga yetganda uning X pozitsiyasi bloklanadi.

/* ============================================================================
 * C99 / Raylib: Super Mario 2D Platformer Core Engine
 * ============================================================================ */
#include <stdbool.h>
#include <math.h>

#define TILE_SIZE             32.0f
#define MAP_WIDTH             100
#define MAP_HEIGHT            15

#define GRAVITY               980.0f
#define FALL_GRAVITY_MULT     1.8f
#define LOW_JUMP_GRAVITY_MULT 2.5f
#define JUMP_IMPULSE          450.0f
#define HORIZONTAL_ACCEL      800.0f
#define FRICTION              1000.0f
#define MAX_RUN_SPEED         220.0f
#define STOMP_BOUNCE_IMPULSE  320.0f

typedef struct { float x, y; } Vec2;
typedef struct { float x, y, width, height; } AABB;

typedef enum { TILE_AIR = 0, TILE_SOLID, TILE_COIN } TileType;
typedef enum { STATE_IDLE, STATE_RUN, STATE_JUMP, STATE_FALL, STATE_DEAD } PlayerState;
typedef enum { ENEMY_PATROL, ENEMY_DEAD } EnemyState;

typedef struct {
    Vec2 position;
    Vec2 velocity;
    Vec2 size;
    PlayerState state;
    bool isGrounded;
    bool isDead;
    int coinsCollected;
} Player;

typedef struct {
    Vec2 position;
    Vec2 velocity;
    Vec2 size;
    EnemyState state;
    int moveDirection;
    bool isActive;
} Enemy;

typedef struct {
    TileType tiles[MAP_HEIGHT][MAP_WIDTH];
} LevelGrid;

typedef struct {
    float x;
    float maxScrollX;
    float viewportWidth;
} OneWayCamera;

static inline bool CheckAABBCollision(AABB a, AABB b) {
    return (a.x < b.x + b.width && a.x + a.width > b.x &&
            a.y < b.y + b.height && a.y + a.height > b.y);
}

static inline float Approach(float current, float target, float delta) {
    return (current < target) ? fminf(current + delta, target) 
                              : fmaxf(current - delta, target);
}

void ResolveMapCollisions(Player *p, LevelGrid *map, float dt) {
    // 1. X Axis
    p->position.x += p->velocity.x * dt;
    AABB boxX = { p->position.x, p->position.y, p->size.x, p->size.y };
    int minX = (int)(boxX.x / TILE_SIZE), maxX = (int)((boxX.x + boxX.width) / TILE_SIZE);
    int minY = (int)(boxX.y / TILE_SIZE), maxY = (int)((boxX.y + boxX.height) / TILE_SIZE);

    for (int ty = minY; ty <= maxY; ty++) {
        for (int tx = minX; tx <= maxX; tx++) {
            if (tx < 0 || tx >= MAP_WIDTH || ty < 0 || ty >= MAP_HEIGHT) continue;
            if (map->tiles[ty][tx] == TILE_SOLID) {
                AABB tileBox = { tx * TILE_SIZE, ty * TILE_SIZE, TILE_SIZE, TILE_SIZE };
                if (CheckAABBCollision(boxX, tileBox)) {
                    if (p->velocity.x > 0) p->position.x = tileBox.x - p->size.x;
                    else if (p->velocity.x < 0) p->position.x = tileBox.x + tileBox.width;
                    p->velocity.x = 0;
                    boxX.x = p->position.x;
                }
            } else if (map->tiles[ty][tx] == TILE_COIN) {
                map->tiles[ty][tx] = TILE_AIR;
                p->coinsCollected++;
            }
        }
    }

    // 2. Y Axis
    p->position.y += p->velocity.y * dt;
    AABB boxY = { p->position.x, p->position.y, p->size.x, p->size.y };
    minX = (int)(boxY.x / TILE_SIZE); maxX = (int)((boxY.x + boxY.width) / TILE_SIZE);
    minY = (int)(boxY.y / TILE_SIZE); maxY = (int)((boxY.y + boxY.height) / TILE_SIZE);
    p->isGrounded = false;

    for (int ty = minY; ty <= maxY; ty++) {
        for (int tx = minX; tx <= maxX; tx++) {
            if (tx < 0 || tx >= MAP_WIDTH || ty < 0 || ty >= MAP_HEIGHT) continue;
            if (map->tiles[ty][tx] == TILE_SOLID) {
                AABB tileBox = { tx * TILE_SIZE, ty * TILE_SIZE, TILE_SIZE, TILE_SIZE };
                if (CheckAABBCollision(boxY, tileBox)) {
                    if (p->velocity.y > 0) {
                        p->position.y = tileBox.y - p->size.y;
                        p->isGrounded = true;
                    } else if (p->velocity.y < 0) {
                        p->position.y = tileBox.y + tileBox.height;
                    }
                    p->velocity.y = 0;
                    boxY.y = p->position.y;
                }
            }
        }
    }
}

void UpdatePlayerKinematics(Player *p, float inputDirX, bool jumpPressed, bool jumpHeld, float dt) {
    if (p->state == STATE_DEAD) {
        p->velocity.y += GRAVITY * dt;
        p->position.y += p->velocity.y * dt;
        return;
    }

    if (inputDirX != 0.0f) {
        p->velocity.x += inputDirX * HORIZONTAL_ACCEL * dt;
        if (p->velocity.x > MAX_RUN_SPEED)  p->velocity.x = MAX_RUN_SPEED;
        if (p->velocity.x < -MAX_RUN_SPEED) p->velocity.x = -MAX_RUN_SPEED;
    } else {
        p->velocity.x = Approach(p->velocity.x, 0.0f, FRICTION * dt);
    }

    if (jumpPressed && p->isGrounded) {
        p->velocity.y = -JUMP_IMPULSE;
        p->isGrounded = false;
    }

    float currentGravity = GRAVITY;
    if (p->velocity.y > 0) currentGravity *= FALL_GRAVITY_MULT;
    else if (p->velocity.y < 0 && !jumpHeld) currentGravity *= LOW_JUMP_GRAVITY_MULT;
    p->velocity.y += currentGravity * dt;

    if (!p->isGrounded) {
        p->state = (p->velocity.y < 0) ? STATE_JUMP : STATE_FALL;
    } else {
        p->state = (fabsf(p->velocity.x) > 5.0f) ? STATE_RUN : STATE_IDLE;
    }
}

void UpdateEnemy(Enemy *e, LevelGrid *map, Player *p, float dt) {
    if (!e->isActive || e->state == ENEMY_DEAD) return;

    e->velocity.x = e->moveDirection * 50.0f;
    e->position.x += e->velocity.x * dt;

    float frontProbeX = (e->moveDirection > 0) ? e->position.x + e->size.x + 2.0f : e->position.x - 2.0f;
    int tileFrontX = (int)(frontProbeX / TILE_SIZE);
    int tileCenterY = (int)((e->position.y + e->size.y * 0.5f) / TILE_SIZE);
    int tileBelowY = (int)((e->position.y + e->size.y + 2.0f) / TILE_SIZE);

    if (map->tiles[tileCenterY][tileFrontX] == TILE_SOLID || map->tiles[tileBelowY][tileFrontX] == TILE_AIR) {
        e->moveDirection *= -1;
    }

    AABB pBox = { p->position.x, p->position.y, p->size.x, p->size.y };
    AABB eBox = { e->position.x, e->position.y, e->size.x, e->size.y };

    if (CheckAABBCollision(pBox, eBox) && p->state != STATE_DEAD) {
        float playerFeet = p->position.y + p->size.y;
        float enemyMidY = e->position.y + (e->size.y * 0.5f);

        if (p->velocity.y > 0 && playerFeet <= enemyMidY + 8.0f) {
            e->state = ENEMY_DEAD;
            e->isActive = false;
            p->velocity.y = -STOMP_BOUNCE_IMPULSE;
        } else {
            p->state = STATE_DEAD;
            p->isDead = true;
            p->velocity.y = -JUMP_IMPULSE * 0.8f;
        }
    }
}

void UpdateOneWayCamera(OneWayCamera *cam, Player *p, float dt) {
    float targetX = p->position.x;
    cam->x += (targetX - cam->x) * 5.0f * dt;

    if (cam->x > cam->maxScrollX) cam->maxScrollX = cam->x;
    else cam->x = cam->maxScrollX;

    float screenLeftEdge = cam->x - (cam->viewportWidth * 0.5f);
    if (p->position.x < screenLeftEdge) {
        p->position.x = screenLeftEdge;
        if (p->velocity.x < 0) p->velocity.x = 0;
    }
}
Mexanika Matematik Formula Tavsif
Chiziqli Tezlanish v_x(t) = clamp(v_x + a · dt, -v_max, v_max) Input bosilganda gorizontal tezlikning ortishi
Inersiya Ishqalanishi v_x(t) = Approach(v_x, 0, μ · dt) Input bo'lmaganda sirpanishni silliq so'ndirish
O'zgaruvchan Gravitatsiya g_eff = g · (v_y > 0 ? 1.8 : (!held ? 2.5 : 1.0)) Mario o'yinlaridagi tushish va sakrash asimmetriyasi
AABB Kesishishi (x₁ < x₂ + w₂) ∧ (x₁ + w₁ > x₂) ∧ (y₁ < y₂ + h₂) ∧ (y₁ + h₁ > y₂) Ikki to'rtburchak to'qnashuvini O(1) vaqtda aniqlash
